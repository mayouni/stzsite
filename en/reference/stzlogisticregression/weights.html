Weights()

No explanation in the source yet: weights

Basic examples (1)

Softanza
o = new stzLogisticRegression(new stzTrainingSet([
		[ [0.0], "lo" ], [ [0.1], "lo" ], [ [0.2], "lo" ],
		[ [0.8], "hi" ], [ [0.9], "hi" ], [ [1.0], "hi" ] ]))
o.Train(300)
# the weight has the sign the data implies
? o.Weights()[1] > 0
# low input -> low class
? o.Classify([0.05])
# high input -> high class
? o.Classify([0.95])
# ...and the boundary sits between them
? o.Probability([0.05]) < 0.5 and o.Probability([0.95]) > 0.5
# it fits its own training data
? o.TrainingAccuracy()
# Why() still narrates
? StzFindFirst("P('hi')", o.Why()) > 0
