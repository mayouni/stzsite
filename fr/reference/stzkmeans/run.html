Référence · stzKMeans

Run()

Pas encore d'explication dans la source : run

Exemples de base (1)

Softanza
oK = new stzKMeans([ [1,1], [1.2,0.9], [8,8], [7.9,8.1] ])
oK.SetK(2)
oK.Run(50)
aC = oK.Clusters()
# two clusters come back
? len(aC)
# ...and the far point lands with the far group
? oK.ClusterOf([7.5, 8.0])
# ...while the near point does not
? oK.ClusterOf([1, 1]) != oK.ClusterOf([8, 8])
oSet = new stzTrainingSet([])
oSet.AddExample([1,1], "near")
oSet.AddExample([1.1,0.9], "near")
oSet.AddExample([9,9], "far")
oSet.AddExample([9.1,8.9], "far")
oKnn = new stzKnn(oSet)
oKnn.SetK(1)
# a query near the origin is classified 'near'
? oKnn.Classify([1.05,1])
# ...and one out at nine is 'far'
? oKnn.Classify([8.9,9])
