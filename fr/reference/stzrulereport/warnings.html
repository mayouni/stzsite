Exemples de base (1)

Softanza
oRep = new stzRuleReport("sev")
oSG = new stzSecurityGraph("s")
oSG.AddSecret("inline")
oSG.AddSite("prod")
oSG.References("prod", "inline")     # unstored-secret -> WARNING only
oRep.Collect(oSG)
# there is a warning
? len(oRep.Warnings()) >= 1
# ...no error
? len(oRep.Errors())
# ...so the gate is SOUND (warnings advise)
? oRep.IsSound()
